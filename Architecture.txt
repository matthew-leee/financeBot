  ## Fold Builder

  def make_temporal_group_folds(
      panel: pd.DataFrame,
      *,
      timestamp_col: str,
      symbol_col: str,
      label_end_col: str,
      n_splits: int,
      min_train_period: timedelta,
      test_period: timedelta,
      embargo: timedelta,
  ) -> list[TemporalFold]:
      """
      Build calendar-time walk-forward folds across all symbols.

      Pseudocode:
          assert required columns exist
          convert timestamp_col and label_end_col to timezone-aware UTC
          sorted_panel = panel.sort_values([timestamp_col, symbol_col])

          first_time = min(sorted_panel[timestamp_col])
          last_time = max(sorted_panel[timestamp_col])
          first_test_start = first_time + min_train_period

          for fold_id in range(n_splits):
              test_start = first_test_start + fold_id * test_period
              test_end = test_start + test_period
              train_end = test_start - embargo

              train_mask =
                  sorted_panel[timestamp_col] < train_end
                  AND sorted_panel[label_end_col] < test_start

              test_mask =
                  sorted_panel[timestamp_col] >= test_start
                  AND sorted_panel[timestamp_col] < test_end

              reject fold if train/test empty

              assert max(train.timestamp) < min(test.timestamp)
              assert max(train.label_end_time) < min(test.timestamp)

              append TemporalFold(...)

      Requirement:
          embargo >= max(label_horizons) for overlapping forward labels.
      """

  ## Validation Runner

  def walk_forward_validate_panel(
      *,
      panel: pd.DataFrame,
      feature_cols: list[str],
      target_col: str,
      timestamp_col: str,
      symbol_col: str,
      label_end_col: str,
      model_factory: Callable[[], PredictiveModel],
      n_splits: int,
      min_train_period: timedelta,
      test_period: timedelta,
      embargo: timedelta,
      transaction_cost_bps: float,
  ) -> PanelWalkForwardReport:
      """
      Run leakage-safe panel validation.

      Pseudocode:
          panel = panel.sort_values([timestamp_col, symbol_col])
          folds = make_temporal_group_folds(...)

          for fold:
              train = panel.loc[fold.train_idx]
              test = panel.loc[fold.test_idx]

              model = model_factory()
              model.fit(train[feature_cols], train[target_col])

              test_scores = model.predict(test[feature_cols])

              for each timestamp in test:
                  rank symbols by score
                  long top quantile
                  optionally underweight/avoid bottom quantile
                  compute paper weights w_t

                  gross_return_t = Σ_i w_{t-1,i} * realized_return_{t,i}
                  turnover_t = Σ_i |w_{t,i} - w_{t-1,i}|
                  cost_t = turnover_t * transaction_cost_bps / 10000
                  net_return_t = gross_return_t - cost_t

              rank_ic = mean Spearman(score, realized forward return) by timestamp
              hit_rate = mean(sign(score) == sign(target))
              net_sharpe = mean(net_returns) / std(net_returns) * sqrt(periods_per_year)
              max_drawdown = min(cumulative_net - cumulative_net.cummax)
              cost_adjusted_pnl = sum(net_returns)

          aggregate mean/std/worst fold metrics
          leakage_checks:
              no_train_after_test_start
              no_label_overlap
              all_symbols_split_by_time
              deterministic_after_shuffle
      """

  ## Covariance Estimation

      def estimate_covariance(
          self,
          returns: pd.DataFrame,
          *,
          halflife_days: int,
          shrinkage: float,
      ) -> pd.DataFrame:
          """
          Estimate PSD covariance matrix.

          EWMA:
              Σ_ewma = E_λ[(r_t - μ)(r_t - μ)^T]

          Shrinkage:
              Σ_shrunk = (1 - α)Σ_ewma + α diag(Σ_ewma)

          PSD floor:
              Σ = Q diag(max(eigenvalue, eigen_floor)) Qᵀ

          Pseudocode:
              returns = returns.sort_index().dropna(how="all")
              ewma_cov = exponentially weighted covariance using halflife_days
              diag_cov = diagonal matrix from ewma_cov
              sigma = (1 - shrinkage) * ewma_cov + shrinkage * diag_cov
              sigma = (sigma + sigma.T) / 2

              eigenvalues, eigenvectors = eigh(sigma)
              eigenvalues = clip(eigenvalues, eigen_floor, inf)
              sigma_psd = eigenvectors @ diag(eigenvalues) @ eigenvectors.T

              return DataFrame(sigma_psd, index=symbols, columns=symbols)
          """

  ## Hedge Math

      def compute_pair_hedge_ratio(
          self,
          *,
          target_symbol: str,
          hedge_symbol: str,
          covariance: pd.DataFrame,
      ) -> float:
          """
          Compute pair hedge ratio.

          Formula:
              h* = - Cov(r_target, r_hedge) / Var(r_hedge)

          If the target has weight w_target:
              hedge_weight = w_target * h*

          For a long inverse ETF:
              Cov(target, inverse) < 0
              therefore h* > 0
              so a positive long inverse-ETF weight offsets target beta.

          Guardrails:
              if Var(hedge) <= epsilon: return 0.0
              clip hedge_weight later by max_hedge_weight
          """

      def compute_portfolio_hedge_overlay(
          self,
          *,
          core_weights: pd.Series,
          covariance: pd.DataFrame,
      ) -> pd.Series:
          """
          Compute multi-instrument minimum-variance hedge overlay.

          Partition symbols:
              C = core symbols
              H = hedge symbols

          Minimum-variance overlay:
              h_raw = - Σ_HH⁻¹ Σ_HC w_C

          Decay/carry sparsity penalty:
              h_j = sign(h_raw_j) * max(
                  0,
                  |h_raw_j| - λ_decay * decay_cost_j / (2 * Σ_HH,jj)
              )

          Hedge effectiveness:
              pre_var = w_Cᵀ Σ_CC w_C
              variance_reduction = w_Cᵀ Σ_CH Σ_HH⁻¹ Σ_HC w_C
              effectiveness = variance_reduction / pre_var

          Accept hedge only if:
              effectiveness >= min_hedge_effectiveness

          Pseudocode:
              core = nonzero core_weights
              hedges = hedge_universe

              sigma_hh = covariance.loc[hedges, hedges]
              sigma_hc = covariance.loc[hedges, core]
              sigma_hh_inv = pinv(psd_floor(sigma_hh))

              h_raw = - sigma_hh_inv @ sigma_hc @ core_weights[core]

              apply soft-threshold decay penalty
              clip each hedge to ±max_hedge_weight
              compute effectiveness

              if pre_var <= epsilon or effectiveness < threshold:
                  return zeros indexed by hedge_universe

              return hedge weights
          """

  ## Core Portfolio Optimization

      def optimize_core_portfolio(
          self,
          *,
          expected_returns: pd.Series,
          covariance: pd.DataFrame,
          previous_weights: pd.Series | None,
      ) -> pd.Series:
          """
          Build constrained core allocation.

          Objective:
              maximize_w:
                  μᵀw
                  - (risk_aversion / 2) * wᵀΣw
                  - turnover_penalty * ||w - w_prev||₁

          Constraints:
              Σ_i |w_i| <= max_gross_exposure
              |Σ_i w_i| <= max_net_exposure
              |w_i| <= max_symbol_weight
              sqrt(wᵀΣw) <= volatility_target
              core shorts disabled unless explicitly enabled

          Initial deterministic implementation:
              volatility_i = sqrt(Σ_ii)
              score_i = μ_i / max(volatility_i, epsilon)
              keep only score_i > 0
              raw_weight_i = score_i / volatility_i
              normalize raw weights
              scale to volatility target
              apply symbol caps
              rescale to gross/net caps
              apply turnover smoothing versus previous_weights
          """

              model_artifacts_valid is False during live trading
              broker_available is False while positions are open
              repeated liquidation failure
